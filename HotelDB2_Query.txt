/* Drop tables first so script can run again without errors */
DROP TABLE IF EXISTS [dbo].[FoodOrders];

DROP TABLE IF EXISTS [dbo].[Users];
GO

/* =========================
   USERS TABLE
========================= */
CREATE TABLE [dbo].[Users](
    [userId] INT IDENTITY(1,1) NOT NULL,
    [userName] VARCHAR(100) NULL,
    [email] VARCHAR(100) NULL,
    [phoneNumber] VARCHAR(20) NULL,
    [role] VARCHAR(50) NULL,
    [status] VARCHAR(50) NULL,
    CONSTRAINT [PK_Users] PRIMARY KEY CLUSTERED ([userId] ASC)
);
GO

/* =========================
   ROOMS TABLE
========================= */
CREATE TABLE [dbo].[Rooms](
    [roomId] INT IDENTITY(1,1) NOT NULL,
    [roomType] VARCHAR(50) NULL,
    [pricePerNight] FLOAT NULL,
    [status] VARCHAR(50) NULL,
    CONSTRAINT [PK_Rooms] PRIMARY KEY CLUSTERED ([roomId] ASC)
);
GO

/* =========================
   BOOKINGS TABLE
========================= */
CREATE TABLE [dbo].[Bookings](
    [bookingId] INT IDENTITY(1,1) NOT NULL,
    [guestId] INT NULL,
    [roomId] INT NULL,
    [checkIn] DATE NULL,
    [checkOut] DATE NULL,
    [status] VARCHAR(50) NULL,
    CONSTRAINT [PK_Bookings] PRIMARY KEY CLUSTERED ([bookingId] ASC),
    CONSTRAINT [FK_Bookings_Users] FOREIGN KEY ([guestId]) REFERENCES [dbo].[Users]([userId]),
    CONSTRAINT [FK_Bookings_Rooms] FOREIGN KEY ([roomId]) REFERENCES [dbo].[Rooms]([roomId])
);
GO

/* =========================
   SERVICES TABLE
========================= */
CREATE TABLE [dbo].[Services](
    [serviceId] INT IDENTITY(1,1) NOT NULL,
    [serviceName] VARCHAR(100) NULL,
    [cost] FLOAT NULL,
    [bookingId] INT NULL,
    [Status] VARCHAR(50) NULL CONSTRAINT [DF_Services_Status] DEFAULT ('Active'),
    CONSTRAINT [PK_Services] PRIMARY KEY CLUSTERED ([serviceId] ASC),
    CONSTRAINT [FK_Services_Bookings] FOREIGN KEY ([bookingId]) REFERENCES [dbo].[Bookings]([bookingId])
);
GO

/* =========================
   FOOD ORDERS TABLE
========================= */
CREATE TABLE [dbo].[FoodOrders](
    [orderId] INT IDENTITY(1,1) NOT NULL,
    [bookingId] INT NULL,
    [foodName] VARCHAR(100) NULL,
    [quantity] INT NULL,
    [totalPrice] FLOAT NULL,
    [status] VARCHAR(50) NULL,
    CONSTRAINT [PK_FoodOrders] PRIMARY KEY CLUSTERED ([orderId] ASC),
    CONSTRAINT [FK_FoodOrders_Bookings] FOREIGN KEY ([bookingId]) REFERENCES [dbo].[Bookings]([bookingId])
);
GO

/* =========================
   INVOICES TABLE
========================= */
CREATE TABLE [dbo].[Invoices](
    [invoiceId] INT IDENTITY(1,1) NOT NULL,
    [bookingId] INT NULL,
    [totalAmount] FLOAT NULL,
    [paymentDate] DATE NULL,
    [paymentMethod] VARCHAR(50) NULL,
    [paymentStatus] VARCHAR(50) NULL,
    CONSTRAINT [PK_Invoices] PRIMARY KEY CLUSTERED ([invoiceId] ASC),
    CONSTRAINT [FK_Invoices_Bookings] FOREIGN KEY ([bookingId]) REFERENCES [dbo].[Bookings]([bookingId])
);
GO

/* =========================
   LOGIN TABLE
========================= */
CREATE TABLE [dbo].[Login](
    [loginId] INT IDENTITY(1,1) NOT NULL,
    [userId] INT NULL,
    [password] VARCHAR(100) NULL,
    [role] VARCHAR(50) NULL,
    [status] BIT NULL,
    CONSTRAINT [PK_Login] PRIMARY KEY CLUSTERED ([loginId] ASC),
    CONSTRAINT [FK_Login_Users] FOREIGN KEY ([userId]) REFERENCES [dbo].[Users]([userId])
);
GO
